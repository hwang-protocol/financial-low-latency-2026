    88	int flle_xdp_mold_itch(struct xdp_md *ctx)
    89	{
    90	    void *data = (void *)(long)ctx->data;
    91	    void *end = (void *)(long)ctx->data_end;
    92	    struct ethhdr *eth = data;
    93	    struct iphdr *ip;
    94	    struct udphdr *udp;
    95	    struct mold_header *mold;
    96	    struct policy_config *cfg;
    97	    struct drop_audit *audit;
    98	    __u8 *lengthp, *itch, *stock;
    99	    __u16 message_length, message_count, locate;
   100	    __u32 zero = 0;
   101	    __u64 now_ns, sequence, old_remaining;
   102	
   103	    if ((void *)(eth + 1) > end || eth->h_proto != bpf_htons(ETH_P_IP))
   104	        return XDP_PASS;
   105	    ip = (void *)(eth + 1);
   106	    if ((void *)(ip + 1) > end || ip->version != 4 || ip->ihl != 5)
   107	        return XDP_PASS;
   108	    if (ip->protocol != IPPROTO_UDP ||
   109	        (ip->frag_off & bpf_htons(IPV4_FRAGMENT_MASK)) != 0 ||
   110	        ip->saddr != bpf_htonl(FEED_SOURCE) ||
   111	        ip->daddr != bpf_htonl(FEED_GROUP))
   112	        return XDP_PASS;
   113	    udp = (void *)(ip + 1);
   114	    if ((void *)(udp + 1) > end || udp->dest != bpf_htons(FEED_PORT))
   115	        return XDP_PASS;
   116	
   117	    count(COURSE_UDP);
   118	    mold = (void *)(udp + 1);
   119	    if ((void *)(mold + 1) > end) {
   120	        count(SHORT_INPUT);
   121	        return XDP_PASS;
   122	    }
   123	    message_count = bpf_ntohs(mold->count);
   124	    if (message_count == 0) {
   125	        count(MOLD_HEARTBEAT);
   126	        return XDP_PASS;
   127	    }
   128	    if (message_count != 1) {
   129	        count(OTHER_ITCH);
   130	        return XDP_PASS;
   131	    }
   132	
   133	    lengthp = (void *)(mold + 1);
   134	    if ((void *)(lengthp + 2) > end) {
   135	        count(SHORT_INPUT);
   136	        return XDP_PASS;
   137	    }
   138	    message_length = ((__u16)lengthp[0] << 8) | lengthp[1];
   139	    itch = lengthp + 2;
   140	    if (message_length != ITCH_ADD_LEN) {
   141	        count(OTHER_ITCH);
   142	        return XDP_PASS;
   143	    }
   144	    if ((void *)(itch + ITCH_ADD_LEN) > end) {
   145	        count(SHORT_INPUT);
   146	        return XDP_PASS;
   147	    }
   148	    if (itch[0] != 'A') {
   149	        count(OTHER_ITCH);
   150	        return XDP_PASS;
   151	    }
   152	    if ((void *)(itch + 3) > end) {
   153	        count(SHORT_INPUT);
   154	        return XDP_PASS;
   155	    }
   156	
   157	    locate = ((__u16)itch[1] << 8) | itch[2];
   158	    stock = itch + 24;
   159	    if ((void *)(stock + 8) > end) {
   160	        count(SHORT_INPUT);
   161	        return XDP_PASS;
   162	    }
   163	    if (is_symbol(stock, "AAPL")) {
   164	        count(AAPL_STOCK);
   165	        return XDP_PASS;
   166	    }
   167	    if (!is_symbol(stock, "NVDA") || locate != NVDA_LOCATE) {
   168	        count(OTHER_ITCH);
   169	        return XDP_PASS;
   170	    }
   171	
   172	    count(NVDA_STOCK);
   173	    cfg = bpf_map_lookup_elem(&policy, &zero);
   174	    if (!cfg || cfg->mode != POLICY_BLOCK || cfg->remaining_drops == 0)
   175	        return XDP_PASS;
   176	    now_ns = bpf_ktime_get_ns();
   177	    if (cfg->deadline_ns && now_ns > cfg->deadline_ns)
   178	        return XDP_PASS;
   179	
   180	    /* 本课策略只允许 1 -> 0。原子比较交换保证多队列并发时也只丢一次。 */
   181	    old_remaining = __sync_val_compare_and_swap(&cfg->remaining_drops, 1, 0);
   182	    if (old_remaining != 1)
   183	        return XDP_PASS;
   184	
   185	    sequence = bpf_be64_to_cpu(mold->sequence);
   186	    audit = bpf_map_lookup_elem(&drop_audit, &zero);
   187	    if (audit) {
   188	        if (!audit->first_drop_sequence)
   189	            audit->first_drop_sequence = sequence;
   190	        audit->first_drop_locate = locate;
   191	        audit->drops += 1;
   192	    }
   193	    count(POLICY_DROP);
   194	    return XDP_DROP;
   195	}
